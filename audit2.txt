import { createClient } from '@supabase/supabase-js';
import { NextResponse } from 'next/server';

const ALLOWED_ROLES = ['personal', 'student', 'teacher'];

function bad(message, status = 400) {
  return NextResponse.json({ error: message }, { status });
}

function getClients() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const anonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
  const serviceKey = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !anonKey) throw new Error('Missing Supabase URL or anon key.');
  if (!serviceKey) throw new Error('Missing SUPABASE_SERVICE_ROLE_KEY.');
  const opts = { auth: { persistSession: false, autoRefreshToken: false } };
  return {
    anon: createClient(url, anonKey, opts),
    admin: createClient(url, serviceKey, opts),
  };
}

export async function POST(request) {
  try {
    const { anon, admin } = getClients();

    let body;
    try {
      body = await request.json();
    } catch {
      return bad('Invalid request.');
    }

    // 1. Validate inputs. The role is checked against a fixed list, never trusted.
    const name = typeof body.name === 'string' ? body.name.trim() : '';
    const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : '';
    const password = typeof body.password === 'string' ? body.password : '';
    const role = body.role;
    const schoolCode = typeof body.schoolCode === 'string' ? body.schoolCode.trim().toUpperCase() : '';
    const specificClassId = body.specificClassId ? String(body.specificClassId).slice(0, 64) : null;
    const documentUrl = typeof body.documentUrl === 'string' ? body.documentUrl.trim() : '';

    if (!ALLOWED_ROLES.includes(role)) return bad('Invalid account type.');
    if (!name || name.length > 80) return bad('Please enter your name.');
    if (!email || email.length > 254 || !email.includes('@')) return bad('Please enter a valid email address.');
    if (password.length < 6 || password.length > 72) return bad('Password must be 6 to 72 characters.');
    if (schoolCode.length > 40) return bad('Invalid school code.');

    // 2. Role-specific checks
    if (role === 'student') {
      if (!schoolCode) return bad('School code is required for student accounts.');

      const { data: codeCheck, error: codeErr } = await admin
        .from('school_classes')
        .select('school_code')
        .eq('school_code', schoolCode)
        .limit(1);
      if (codeErr || !codeCheck || codeCheck.length === 0) {
        return bad('Invalid classroom clearance token.');
      }

      if (specificClassId) {
        const { data: classCheck, error: classErr } = await admin
          .from('school_classes')
          .select('id')
          .eq('id', specificClassId)
          .eq('school_code', schoolCode)
          .limit(1);
        if (classErr || !classCheck || classCheck.length === 0) {
          return bad('That classroom does not belong to this school.');
        }
      }
    }

    if (role === 'teacher') {
      if (!schoolCode) return bad('School code is required for admin accounts.');
      if (!documentUrl || documentUrl.length > 500 || !documentUrl.startsWith('https://')) {
        return bad('An appointment letter is required for admin accounts.');
      }
    }

    // 3. Create the auth user
    const { data: authData, error: authError } = await anon.auth.signUp({ email, password });
    if (authError) return bad(authError.message);
    if (!authData?.user) return bad('Could not create the account.', 500);

    // With email confirmation on, an existing email comes back as a user with no identities.
    if (Array.isArray(authData.user.identities) && authData.user.identities.length === 0) {
      return bad('An account with this email already exists.');
    }

    const userId = authData.user.id;

    // 4. Create the profile with the server-only key. Everything privileged is set here.
    const initialBalance = 50000;
    const usesSchool = role !== 'personal';

    const { error: profileError } = await admin.from('profiles').insert([{
      id: userId,
      name,
      email,
      role,
      wallet_balance: initialBalance,
      net_worth: initialBalance,
      school_code: usesSchool ? schoolCode : null,
      school_id: usesSchool ? schoolCode : null,
      specific_class_id: role === 'student' && specificClassId ? specificClassId : null,
      verification_status: role === 'teacher' ? 'pending' : 'approved',
      verification_document_url: role === 'teacher' ? documentUrl : null,
      student_approved: true
    }]);

    if (profileError) {
      console.error('Signup: profile insert failed', profileError);
      // Do not leave an account with no profile behind.
      await admin.auth.admin.deleteUser(userId);
      return bad('Could not finish creating your account. Please try again.', 500);
    }

    return NextResponse.json({ success: true, user: userId });
  } catch (error) {
    console.error('Signup route error', error);
    return bad('Signup is temporarily unavailable. Please try again.', 500);
  }
}

function SignupForm() {

  const [isLogin, setIsLogin] = useState(false);

  // Step 1: Personal or School. Step 2 (School only): Student or Admin.
  // Admin is stored as the existing 'teacher' role so approvals and the admin queue keep working.
  const [accountGroup, setAccountGroup] = useState('personal');
  const [schoolRole, setSchoolRole] = useState('student');
  const accountType = accountGroup === 'personal' ? 'personal' : schoolRole;

  const [documentFile, setDocumentFile] = useState(null);
  const [availableClasses, setAvailableClasses] = useState([]);
  const [formData, setFormData] = useState({ name: '', email: '', password: '', schoolCode: '', specificClassId: '' });
  const [status, setStatus] = useState({ loading: false, message: '', success: false });

  // Robot Interactive States
  const [mood, setMood] = useState('idle');
  const [isTurned, setIsTurned] = useState(false);
  const [speechText, setSpeechText] = useState("Greetings trader. I am Volt, guardian of Bull Run.");
  const [eyePos, setEyePos] = useState({ x: 0, y: 0 });
  const [headTilt, setHeadTilt] = useState({ rx: 0, ry: 0 });
  const [passScore, setPassScore] = useState(0);

  // Forgot Password System States
  const [isForgotMode, setIsForgotMode] = useState(false);
  const [maskedEmail, setMaskedEmail] = useState('');
  const [targetEmail, setTargetEmail] = useState('');
  const [confirmEmailInput, setConfirmEmailInput] = useState('');

  useEffect(() => {
    setIsLogin(searchParams.get('mode') === 'login');
    setIsForgotMode(false);
  }, [searchParams]);

  const say = (text) => setSpeechText(text);

  const chooseGroup = (group) => {
    setAccountGroup(group);

    if (group === 'personal') {
      setFormData((prev) => ({ ...prev, schoolCode: '', specificClassId: '' }));
      setAvailableClasses([]);
      setDocumentFile(null);
      say('Personal track selected. Just you and the market.');
    } else {
      say('School track! Are you a student or an admin?');
    }
  };

  const chooseSchoolRole = (role) => {
    setSchoolRole(role);
    setFormData((prev) => ({ ...prev, specificClassId: '' }));
    if (role !== 'teacher') setDocumentFile(null);
    say(role === 'student' ? 'Student track. Enter your school code.' : 'Admin track. I will need your appointment letter.');

    const ratio = Math.min(valLength / 22, 1);

    setHeadTilt({ rx: -8, ry: -5 + 10 * ratio });
  };

  const handlePasswordInput = (val) => {
    setFormData((prev) => ({ ...prev, password: val }));
    let score = 0;
    if (val.length >= 8) score++;
    if (/[a-z]/.test(val) && /[A-Z]/.test(val)) score++;
    if (/\d/.test(val)) score++;
    if (/[^a-zA-Z0-9]/.test(val)) score++;
    if (val.length > 0 && score === 0) score = 1;
    setPassScore(score);
  };

  const handleSchoolCodeBlur = async (code) => {
    if (!code) return;
    const { data, error } = await supabase
      .from('school_classes')
      .select('id, class_name')
      .eq('school_code', code.toUpperCase());
    
    if (!error && data) {

  };

  const initForgotPasswordFlow = () => {
    const rawEmail = formData.email.trim();
    if (!rawEmail || !rawEmail.includes('@')) {
      setStatus({ loading: false, message: 'Please input your email address above first.', success: false });
      say("Type your email address above so I can compute the security mask.");
      return;
    }

    const [username, domain] = rawEmail.split('@');
    if (username.length <= 2) {
      setMaskedEmail(`${username[0]}***@${domain}`);
    } else {
      setMaskedEmail(`${username[0]}${'*'.repeat(username.length - 2)}${username[username.length - 1]}@${domain}`);

    e.preventDefault();
    setStatus({ loading: true, message: '', success: false });

      setStatus({ loading: false, message: 'The confirmation email handle does not match.', success: false });
      say("Access denied! Email handle mismatch detected.");
      return;
    }

      // Safely resolve redirect origin across environments

      const redirectUrl = `${redirectOrigin}/auth/update-password`;

      const { data, error } = await supabase.auth.resetPasswordForEmail(targetEmail, {
        redirectTo: redirectUrl,

// SERVER ONLY. Never prefix this key with NEXT_PUBLIC_.
export const admin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY,
  { auth: { persistSession: false, autoRefreshToken: false } }
);

export async function getUserFromRequest(req) {
  const token = (req.headers.get('authorization') || '').replace(/^Bearer\s+/i, '');
  if (!token) return null;
  const { data, error } = await admin.auth.getUser(token);
  return error ? null : data?.user ?? null;
}

import { admin, getUserFromRequest } from '@/lib/supabaseAdmin';
import { getServerPrices, recomputeNetWorth } from '@/lib/serverMarket';

export const dynamic = 'force-dynamic';

export async function POST(req) {
  try {
    const user = await getUserFromRequest(req);
    if (!user) return Response.json({ error: 'Unauthorized' }, { status: 401 });

    const { sym, side, qty } = await req.json();
    const quantity = Number(qty);

    if (typeof sym !== 'string' || !['BUY', 'SELL'].includes(side) ||
        !Number.isFinite(quantity) || quantity <= 0 || quantity > 1e6) {
      return Response.json({ error: 'Invalid order' }, { status: 400 });
    }

    const prices = await getServerPrices(new URL(req.url).origin);
    const price = prices[sym];
    if (!price) return Response.json({ error: 'Price not available' }, { status: 400 });

    const { data, error } = await admin.rpc('execute_trade', {
      p_user: user.id, p_sym: sym, p_side: side, p_qty: quantity, p_price: price,
    });
    if (error) return Response.json({ error: error.message }, { status: 400 });

    const netWorth = await recomputeNetWorth(user.id, prices);
    return Response.json({ balance: data.balance, price, netWorth });
  } catch (e) {
    console.error('trade failed', e);
    return Response.json({ error: 'Trade failed' }, { status: 500 });
  }
}
