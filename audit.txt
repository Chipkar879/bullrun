'use client';

import { useState, useEffect, useMemo, useCallback, useRef } from 'react';
import Navbar from '@/components/Navbar';
import { supabase } from '@/lib/supabase';

// âš ï¸ SUPER ADMIN IMMUNITY EMAIL
const SUPER_ADMIN_EMAIL = 'verymystery18@gmail.com';
const REFRESH_MS = 10000;

const inr = (value) =>
  `â‚¹${Number(value || 0).toLocaleString('en-IN', { maximumFractionDigits: 2 })}`;

// Same value the leaderboard ranks on (cash + holdings, saved by the simulator)
const getNetWorth = (p) => Number(p.net_worth ?? p.wallet_balance ?? 0) || 0;

const formatDuration = (seconds) => {
  const total = Math.max(0, Math.round(Number(seconds) || 0));
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  return h + 'h ' + m + 'm';
};

// Stable order: highest net worth first, ties broken by id (same as the leaderboard)
const rankProfiles = (list) =>
  [...list]
    .sort((a, b) => {
      const diff = getNetWorth(b) - getNetWorth(a);
      if (diff !== 0) return diff;
      return String(a.id).localeCompare(String(b.id));
    })
    .map((p, index) => ({ ...p, rank: index + 1 }));

// HELPER: CALCULATE EXACT EXPIRY & FREEZE STATUS
// `now` is passed in so the UI can re-evaluate every second.
const getProfileFreezeStatus = (p, now = Date.now()) => {
  if (p.email === SUPER_ADMIN_EMAIL) {
    return { isFrozen: false, isSuperAdmin: true, isPermanent: true, expiryMs: null };
  }

  // 1. Explicit admin freeze flag in database ALWAYS takes top priority
  if (p.is_frozen === true) {
    return { isFrozen: true, isSuperAdmin: false, isPermanent: false, expiryMs: 0 };
  }

  // 2. Premium with NO deadline = permanent access
  if (!p.access_expires_at && p.is_premium) {
    return { isFrozen: false, isSuperAdmin: false, isPermanent: true, expiryMs: null };
  }

  // 3. The deadline decides, even for premium / admin-granted accounts
  let expiryMs;
  if (p.access_expires_at) {
    expiryMs = new Date(p.access_expires_at).getTime();
  } else {
    const createdAt = p.created_at ? new Date(p.created_at).getTime() : now;
    expiryMs = createdAt + (7 * 24 * 60 * 60 * 1000);
  }

  return {
    isFrozen: now >= expiryMs,
    isSuperAdmin: false,
    isPermanent: false,
    expiryMs
  };
};

export default function SuperAdminDashboard() {

  const [loading, setLoading] = useState(true);
  const [allProfiles, setAllProfiles] = useState([]);
  const [lastUpdated, setLastUpdated] = useState(null);
  const [activity, setActivity] = useState({});

  // Global clock: makes cards flip to FROZEN the moment time runs out
  const [now, setNow] = useState(Date.now());
  // Remembers which accounts we already auto-froze in the database
  const autoFrozenRef = useRef(new Set());

  // Search & School Category Filter States
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedSchoolFilter, setSelectedSchoolFilter] = useState('ALL');

  // Modal Inspection & Custom Days State
  const [selectedId, setSelectedId] = useState(null);
  const [customDaysInput, setCustomDaysInput] = useState('30');

  // Modal Real-Time Live Countdown State
  const [modalTimeLeft, setModalTimeLeft] = useState({
    days: 0,
    hours: 0,
    minutes: 0,
    seconds: 0,
    isExpired: false
  });

  // The open dossier always reflects the latest synced data
  const selectedProfile = useMemo(
    () => allProfiles.find((p) => p.id === selectedId) || null,
    [allProfiles, selectedId]
  );

  const pendingTeachers = useMemo(
    () => allProfiles.filter((p) => p.role === 'teacher' && p.verification_status === 'pending'),
    [allProfiles]
  );

  const patchProfile = useCallback((profileId, fields) => {
    setAllProfiles((prev) =>
      rankProfiles(prev.map((p) => (p.id === profileId ? { ...p, ...fields } : p)))
    );
  }, []);

    if (!silent) setLoading(true);

    const { data, error } = await supabase.from('profiles').select('*');

    if (error) {
      console.error('Admin profile fetch failed:', error);
      if (!silent) setLoading(false);
      return;
    }

    setAllProfiles(rankProfiles(data || []));
    setLastUpdated(new Date());
    setLoading(false);

    // Weekly time on site: last 7 IST days including today
    const cutoff = new Date(Date.now() + 5.5 * 3600 * 1000 - 6 * 86400 * 1000)
      .toISOString()
      .slice(0, 10);

    const { data: rows, error: actError } = await supabase
      .from('user_activity')
      .select('user_id, seconds')
      .gte('day', cutoff);

    if (actError) {
      console.warn('Activity fetch failed:', actError.message);
      return;

    (rows || []).forEach((r) => {
      totals[r.user_id] = (totals[r.user_id] || 0) + Number(r.seconds || 0);
    });

    const verifyIdentity = async () => {

        setAuthorized(true);
      } else {
        window.location.href = '/';
      }
    };
    verifyIdentity();

    if (!authorized) return undefined;

    fetchMasterData(false);

    const refresh = () => {
      if (!document.hidden) fetchMasterData(true);
    };

    const channel = supabase
      .channel('admin-profiles-live')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'profiles' }, refresh)
      .subscribe();

    const interval = setInterval(refresh, REFRESH_MS);
    document.addEventListener('visibilitychange', refresh);

    return () => {
      clearInterval(interval);
      document.removeEventListener('visibilitychange', refresh);
      supabase.removeChannel(channel);
    };
  }, [authorized, fetchMasterData]);

    const t = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(t);
  }, []);

      const { isFrozen, isSuperAdmin } = getProfileFreezeStatus(p, now);

      // If the account is active again (e.g. admin granted days), allow future auto-freeze
      if (!isFrozen) autoFrozenRef.current.delete(p.id);

      if (!isFrozen || isSuperAdmin || p.is_frozen === true) return;
      if (autoFrozenRef.current.has(p.id)) return;

      autoFrozenRef.current.add(p.id);

      const fields = { is_frozen: true, is_premium: false };
      const { error } = await supabase.from('profiles').update(fields).eq('id', p.id);

      if (error) {
        console.warn('Auto-freeze failed:', error.message);
        return;

      patchProfile(p.id, fields);

        return;

        return;

      const minutes = Math.floor((diffMs / 1000 / 60) % 60);
      const seconds = Math.floor((diffMs / 1000) % 60);

      setModalTimeLeft({ days, hours, minutes, seconds, isExpired: false });

        .from('profiles')
        .update(updatedFields)
        .eq('id', profileId);

  if (loading) {
    return (
      <main className="min-h-screen bg-bull-bg text-slate-100 antialiased font-sans relative">
        <Navbar />
        <div className="max-w-6xl mx-auto px-6 pt-32 pb-24 flex justify-center">
          <div className="space-y-4 text-center">
            <div className="w-12 h-12 border-4 border-bull-lime border-t-transparent rounded-full animate-spin mx-auto" />
            <p className="text-slate-400 text-xs font-bold uppercase tracking-wider animate-pulse font-mono">
              Securing Admin Firewall Layer...
            </p>
          </div>
        </div>
      </main>
    );
  }

  if (!authorized) return null;

  return (
    <main className="min-h-screen bg-bull-bg text-slate-100 antialiased font-sans relative max-w-full overflow-x-hidden pt-24 pb-20">
      <Navbar />

      <section className="max-w-6xl mx-auto px-4 sm:px-6 space-y-8 animate-fadeInFast">

        {/* HEADER BLOCK */}
        <div className="border-b border-bull-line pb-4 flex flex-col md:flex-row md:items-center justify-between gap-4">
          <div>
            <span className="bg-bull-raised text-bull-lime border border-bull-line text-[10px] uppercase tracking-widest font-black px-3 py-1 rounded-full shadow-inner mb-2 inline-block font-mono">
              ðŸ›¡ï¸ Super Admin Clearance

              return (

// Volume pricing: every student is charged at the rate of their band (like tax slabs).
// Bigger schools get a lower rate on the extra students, so the total never drops as size grows.
const SLABS = [
  { from: 1, to: 10, rate: 99 },
  { from: 11, to: 30, rate: 85 },
  { from: 31, to: 100, rate: 72 },
  { from: 101, to: 200, rate: 62 },
  { from: 201, to: MAX_SELF_SERVE, rate: 55 },
];

function schoolPrice(n) {
  let total = 0;
  for (const s of SLABS) {
    if (n >= s.from) {
      const count = Math.min(n, s.to) - s.from + 1;
      total += count * s.rate;
    }
  }
  return total;
}

import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';

const SUPER_ADMIN_EMAIL = 'verymystery18@gmail.com';

export async function middleware(req) {
  const res = NextResponse.next();
  const { pathname } = req.nextUrl;

  // 1. Static files & internal assets always pass
  if (
    pathname.startsWith('/_next') ||
    pathname.startsWith('/api') ||
    pathname.startsWith('/favicon.ico') ||
    pathname === '/icon.svg' ||
    pathname === '/apple-icon.png' ||
    pathname.startsWith('/opengraph-image') ||
    pathname.startsWith('/twitter-image') ||
    pathname === '/sitemap.xml' ||
    pathname === '/robots.txt' ||
    pathname === '/manifest.webmanifest'
  ) {
    return res;
  }

  // 2. Open public pages
  const isAlwaysAllowedPage =
    pathname === '/' ||
    pathname.startsWith('/pricing') ||
    pathname.startsWith('/signup') ||
    pathname.startsWith('/auth');

  // 3. Extract the auth token from cookies
  const allCookies = req.cookies.getAll();
  const authCookie = allCookies.find((c) =>
    c.name.includes('-auth-token') || c.name.startsWith('sb-')
  );

  let user = null;
  let accessToken = null;

  if (authCookie) {
    try {
      let rawVal = authCookie.value;
      if (rawVal.startsWith('base64-')) {
        rawVal = Buffer.from(rawVal.replace('base64-', ''), 'base64').toString('utf-8');
      }
      const parsed = JSON.parse(rawVal);
      accessToken = Array.isArray(parsed) ? parsed[0] : parsed?.access_token || parsed;

      if (accessToken && typeof accessToken === 'string') {
        const supabaseAuth = createClient(
          process.env.NEXT_PUBLIC_SUPABASE_URL,
          process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY
        );
        const { data } = await supabaseAuth.auth.getUser(accessToken);
        user = data?.user || null;
      }
    } catch {
      user = null;
    }
  }

  // NOT LOGGED IN
  if (!user) {
    if (isAlwaysAllowedPage) return res;
    const loginUrl = req.nextUrl.clone();
    loginUrl.pathname = '/signup';
    loginUrl.searchParams.set('mode', 'login');
    return NextResponse.redirect(loginUrl);
  }

  // SUPER ADMIN IMMUNITY
  if (user.email === SUPER_ADMIN_EMAIL) {
    return res;
  }

  // QUERY PROFILE WITH USER'S AUTH HEADER (Bypasses RLS blocks)
  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY,
    {
      global: {
        headers: {
          Authorization: `Bearer ${accessToken}`,
        },
      },
    }
  );

  const { data: profile } = await supabase
    .from('profiles')
    .select('is_frozen, is_premium, access_expires_at, created_at, role')
    .eq('id', user.id)
    .maybeSingle();

  if (profile) {
    if (profile.role === 'admin') return res;

    let expiryMs;
    if (profile.access_expires_at) {
      expiryMs = new Date(profile.access_expires_at).getTime();
    } else {
      const createdAtMs = profile.created_at ? new Date(profile.created_at).getTime() : Date.now();
      expiryMs = createdAtMs + 7 * 24 * 60 * 60 * 1000;
    }

    const hasPremium = Boolean(profile.is_premium);
    const isTrialExpired = Date.now() >= expiryMs && !hasPremium;
    
    // HARD ENFORCEMENT: If is_frozen is TRUE in Supabase, BLOCK IMMEDIATELY
    const isFrozen = Boolean(profile.is_frozen) || isTrialExpired;

    if (isFrozen) {
      const isAllowedForFrozenUser = pathname === '/' || pathname.startsWith('/pricing');
      if (!isAllowedForFrozenUser) {
        const pricingUrl = req.nextUrl.clone();
        pricingUrl.pathname = '/pricing';
        return NextResponse.redirect(pricingUrl);
      }
    }
  }

  return res;
}

        // Check cookie first, fallback to local storage

    pathname.startsWith('/api') ||
    pathname.startsWith('/favicon.ico') ||
    pathname === '/icon.svg' ||
    pathname === '/apple-icon.png' ||
    pathname.startsWith('/opengraph-image') ||
    pathname.startsWith('/twitter-image') ||
    pathname === '/sitemap.xml' ||
    pathname === '/robots.txt' ||
    pathname === '/manifest.webmanifest'
  ) {
    return res;
  }

  // 2. Open public pages
  const isAlwaysAllowedPage =
    pathname === '/' ||
    pathname.startsWith('/pricing') ||
    pathname.startsWith('/signup') ||
    pathname.startsWith('/auth');

  // 3. Extract the auth token from cookies
  const allCookies = req.cookies.getAll();
  const authCookie = allCookies.find((c) =>
    c.name.includes('-auth-token') || c.name.startsWith('sb-')
  );

  let user = null;
  let accessToken = null;

  if (authCookie) {
    try {
      let rawVal = authCookie.value;
      if (rawVal.startsWith('base64-')) {
        rawVal = Buffer.from(rawVal.replace('base64-', ''), 'base64').toString('utf-8');
      }
      const parsed = JSON.parse(rawVal);
      accessToken = Array.isArray(parsed) ? parsed[0] : parsed?.access_token || parsed;

      if (accessToken && typeof accessToken === 'string') {
        const supabaseAuth = createClient(
          process.env.NEXT_PUBLIC_SUPABASE_URL,
          process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY
        );
        const { data } = await supabaseAuth.auth.getUser(accessToken);
        user = data?.user || null;
      }
    } catch {
      user = null;
    }
  }

  // NOT LOGGED IN
  if (!user) {
    if (isAlwaysAllowedPage) return res;
    const loginUrl = req.nextUrl.clone();
    loginUrl.pathname = '/signup';
    loginUrl.searchParams.set('mode', 'login');
    return NextResponse.redirect(loginUrl);
  }

  // SUPER ADMIN IMMUNITY
  if (user.email === SUPER_ADMIN_EMAIL) {
    return res;
  }

  // QUERY PROFILE WITH USER'S AUTH HEADER (Bypasses RLS blocks)
  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY,
    {
      global: {
        headers: {
          Authorization: `Bearer ${accessToken}`,
        },
      },
    }
  );

  const { data: profile } = await supabase
    .from('profiles')
    .select('is_frozen, is_premium, access_expires_at, created_at, role')
    .eq('id', user.id)
    .maybeSingle();

  if (profile) {
    if (profile.role === 'admin') return res;

    let expiryMs;
    if (profile.access_expires_at) {
      expiryMs = new Date(profile.access_expires_at).getTime();
    } else {
      const createdAtMs = profile.created_at ? new Date(profile.created_at).getTime() : Date.now();
      expiryMs = createdAtMs + 7 * 24 * 60 * 60 * 1000;
    }

    const hasPremium = Boolean(profile.is_premium);
    const isTrialExpired = Date.now() >= expiryMs && !hasPremium;
    
    // HARD ENFORCEMENT: If is_frozen is TRUE in Supabase, BLOCK IMMEDIATELY
    const isFrozen = Boolean(profile.is_frozen) || isTrialExpired;

    if (isFrozen) {
      const isAllowedForFrozenUser = pathname === '/' || pathname.startsWith('/pricing');
      if (!isAllowedForFrozenUser) {
        const pricingUrl = req.nextUrl.clone();
        pricingUrl.pathname = '/pricing';
        return NextResponse.redirect(pricingUrl);
      }
    }
  }

  return res;
}
